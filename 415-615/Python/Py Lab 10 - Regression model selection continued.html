# ================================================================
# PART 4. SEQUENTIAL F-TESTS
# ================================================================

# Sequential selection procedures add or remove predictors
# one at a time, using partial F-tests.
#
# Unlike the AIC-based selection in Lab 9, the decision
# to add or remove a predictor depends on its p-value.
#
# We specify a significance level alpha.
#
# FORWARD SELECTION:
#   Start with the intercept-only model.
#   At each step, test all excluded predictors.
#   Add the predictor with the smallest partial F-test p-value,
#   provided that p-value is below alpha.
#   Stop when no excluded predictor meets the criterion.
#
# BACKWARD ELIMINATION:
#   Start with the full model.
#   At each step, test all included predictors.
#   Remove the predictor with the largest partial F-test
#   p-value, provided that p-value exceeds alpha.
#   Stop when all remaining predictors meet the criterion.
#
# The procedures use partial F-tests, comparing nested
# regression models.
#
# For a single predictor, the partial F-test is equivalent
# to the usual t-test for its regression coefficient.


# ------------------------------------------------
# 4A. Define the response and candidate predictors
# ------------------------------------------------

Y = longley["Unemployed"]

predictors = [
    "GNP.deflator",
    "GNP",
    "Armed.Forces",
    "Population",
    "Year",
    "Employed"
]


# Define a function to fit a regression model using
# any subset of these predictors.

def fit_model(selected):

    X = sm.add_constant(
        longley[selected],
        has_constant="add"
    )

    return sm.OLS(Y, X).fit()


# ------------------------------------------------
# 4B. Define the sequential F-test procedure
# ------------------------------------------------

# The following function performs forward selection
# or backward elimination.
#
# direction:
#   "forward"  - add significant predictors
#   "backward" - remove insignificant predictors
#
# alpha:
#   Significance level for the partial F-tests.
#
# correction:
#   If True, apply a stepwise Bonferroni adjustment
#   by dividing alpha by the number of candidate tests
#   performed at the current step.
#
# trace:
#   If True, display every selection step.
#
# The function returns the final fitted regression model.


def select_F(direction="forward", alpha=0.05,
             correction=False, trace=False):

    if direction not in ("forward", "backward"):
        raise ValueError("direction must be 'forward' or 'backward'")

    # Forward selection starts with no predictors.
    # Backward elimination starts with all predictors.

    selected = (
        [] if direction == "forward"
        else predictors.copy()
    )

    step = 0

    while True:

        current = fit_model(selected)

        # ----------------------------------------
        # Forward selection
        # ----------------------------------------

        if direction == "forward":

            remaining = [
                v for v in predictors if v not in selected
            ]

            if not remaining:
                break

            # Compare each enlarged model with the
            # current model using a partial F-test.

            tests = []

            for v in remaining:

                enlarged = fit_model(selected + [v])

                F, p, df = enlarged.compare_f_test(current)

                tests.append((v, F, p))

            # Select the predictor with the smallest p-value.

            variable, F, p = min(
                tests,
                key=lambda x: x[2]
            )

            # Bonferroni adjustment, if requested.

            cutoff = (
                alpha / len(tests)
                if correction else alpha
            )

            if p >= cutoff:
                break

            selected.append(variable)

            action = "Added"


        # ----------------------------------------
        # Backward elimination
        # ----------------------------------------

        else:

            if not selected:
                break

            # Compare the current model with each model
            # obtained by removing one predictor.

            tests = []

            for v in selected:

                reduced = fit_model(
                    [x for x in selected if x != v]
                )

                F, p, df = current.compare_f_test(reduced)

                tests.append((v, F, p))

            # Select the predictor with the largest p-value.

            variable, F, p = max(
                tests,
                key=lambda x: x[2]
            )

            cutoff = (
                alpha / len(tests)
                if correction else alpha
            )

            if p <= cutoff:
                break

            selected.remove(variable)

            action = "Removed"


        # Display the current selection step.

        step += 1

        if trace:

            print(
                f"Step {step}: {action} {variable:15s}"
                f"  F = {F:.3f}"
                f"  p = {p:.4f}"
                f"  cutoff = {cutoff:.4f}"
            )

    # Fit and return the final model.

    return fit_model(selected)


# ------------------------------------------------
# 4C. Forward selection at alpha = 0.05
# ------------------------------------------------

# Start with the intercept-only model.
#
# A predictor enters the model if its partial F-test
# p-value is less than 0.05.

reg_forward = select_F(
    direction="forward",
    alpha=0.05
)

print("\nFORWARD SELECTION: alpha = 0.05")

print("Selected predictors:")
print(reg_forward.model.exog_names)

print("\nRegression coefficients:")
print(reg_forward.params)


# ------------------------------------------------
# 4D. Forward selection at alpha = 0.20
# ------------------------------------------------

# Increasing alpha makes it easier for a predictor
# to enter the model.
#
# Consequently, larger models may be selected.

reg_forward20 = select_F(
    direction="forward",
    alpha=0.20
)

print("\nFORWARD SELECTION: alpha = 0.20")

print("Selected predictors:")
print(reg_forward20.model.exog_names)

print("\nRegression coefficients:")
print(reg_forward20.params)


# ------------------------------------------------
# 4E. Forward selection at alpha = 0.001
# ------------------------------------------------

# Decreasing alpha makes the criterion for entering
# the model much more stringent.
#
# The selected model may contain very few predictors
# or even only an intercept.

reg_forward001 = select_F(
    direction="forward",
    alpha=0.001
)

print("\nFORWARD SELECTION: alpha = 0.001")

print("Selected predictors:")
print(reg_forward001.model.exog_names)

print("\nRegression coefficients:")
print(reg_forward001.params)


# ------------------------------------------------
# 4F. Backward elimination
# ------------------------------------------------

# Backward elimination starts with the full model.
#
# At each step, remove the predictor with the largest
# partial F-test p-value if it exceeds alpha = 0.05.

reg_backward = select_F(
    direction="backward",
    alpha=0.05
)

print("\nBACKWARD ELIMINATION: alpha = 0.05")

print("Selected predictors:")
print(reg_backward.model.exog_names)

print("\nRegression coefficients:")
print(reg_backward.params)


# ------------------------------------------------
# 4G. Trace the backward elimination procedure
# ------------------------------------------------

# The trace option displays each removed predictor,
# its partial F-statistic, and its p-value.
#
# This allows us to see how the regression model
# changes during variable selection.

reg_trace = select_F(
    direction="backward",
    alpha=0.05,
    trace=True
)

print("\nFINAL MODEL AFTER BACKWARD ELIMINATION")

print(reg_trace.summary())


# ------------------------------------------------
# 4H. Bonferroni correction
# ------------------------------------------------

# Sequential variable selection involves multiple
# hypothesis tests.
#
# When many tests are performed, the probability of
# at least one false rejection may increase.
#
# The Bonferroni correction reduces the significance
# level used for each individual test.
#
# At each selection step, we use:
#
#     adjusted alpha = alpha / m
#
# where m is the number of candidate tests at that step.
#
# This makes it harder for predictors to enter a model.
#
# IMPORTANT:
# This is a stepwise Bonferroni adjustment, not a
# guarantee of familywise error control over the
# entire adaptive model-selection procedure.


# Forward selection without Bonferroni correction.

reg_no_bonf = select_F(
    direction="forward",
    alpha=0.20
)

print("\nFORWARD SELECTION WITHOUT BONFERRONI")

print(reg_no_bonf.model.exog_names)


# Forward selection with Bonferroni correction.

reg_bonf = select_F(
    direction="forward",
    alpha=0.20,
    correction=True
)

print("\nFORWARD SELECTION WITH BONFERRONI")

print(reg_bonf.model.exog_names)


# ------------------------------------------------
# 4I. Backward elimination with Bonferroni correction
# ------------------------------------------------

# We can also apply the adjustment to backward
# elimination.
#
# Notice that the consequences may differ from
# forward selection: a smaller significance cutoff
# makes it easier to classify an included predictor
# as insufficiently significant and remove it.

reg_backward_bonf = select_F(
    direction="backward",
    alpha=0.05,
    correction=True
)

print("\nBACKWARD ELIMINATION WITH BONFERRONI")

print(reg_backward_bonf.model.exog_names)

# ================================================================
# PART 5. VISUALIZATION: SCATTERPLOT MATRIX
# ================================================================

# A scatterplot matrix displays scatterplots for
# every pair of variables in the dataset.
#
# It is useful for examining:
#
#   1. Relationships between the response and predictors.
#   2. Whether these relationships appear linear.
#   3. Possible multicollinearity among predictors.
#
# The Longley dataset is well known for strong
# multicollinearity among its economic variables.
#
# This makes it a useful example for studying
# regression model selection.


# ------------------------------------------------
# 5A. Create the scatterplot matrix
# ------------------------------------------------

# pandas provides scatter_matrix(), which is similar
# to the plot(longley) command in R.

axes = scatter_matrix(
    longley,
    figsize=(12, 12),
    diagonal="hist",
    alpha=0.8
)

plt.suptitle(
    "Longley Data: Scatterplot Matrix",
    fontsize=15
)

plt.tight_layout()
plt.show()


# ------------------------------------------------
# 5B. Examine correlations
# ------------------------------------------------

# A correlation matrix provides a numerical summary
# of the linear relationships visible in the plots.
#
# Correlations close to +1 or -1 indicate strong
# linear relationships between pairs of variables.
#
# However, multicollinearity can also involve
# combinations of more than two predictors.

print("\nCORRELATION MATRIX")

print(longley.corr().round(3))
