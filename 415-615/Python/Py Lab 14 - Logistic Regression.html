# ================================================================
# PART 2. FITTING A LOGISTIC REGRESSION MODEL
# ================================================================

# Logistic regression is used when the response
# variable is binary.
#
# Let:
#
#     p = P(Diagnosis = 1 | predictors)
#
# The logistic regression model is:
#
#     log(p / (1-p))
#         = b0 + b1*GenderMale
#              + b2*Guardian_status
#              + b3*Cohesion_score
#
# The left-hand side is called the logit.
#
# Logistic regression models the log-odds of a
# positive diagnosis as a linear function of
# the predictors.
#
# Unlike ordinary linear regression, the fitted
# probabilities are constrained to lie between
# 0 and 1.


# ------------------------------------------------
# 2A. Fit the logistic regression
# ------------------------------------------------

# C(Gender, Treatment(reference="Female")) specifies
# that Gender is categorical and Female is the
# reference category.
#
# Guardian_status is treated as a numerical predictor,
# as in the original R lab.
#
# statsmodels uses maximum likelihood estimation
# to estimate the regression coefficients.


fit = smf.logit(
    'Diagnosis ~ C(Gender, Treatment(reference="Female"))'
    ' + Guardian_status + Cohesion_score',
    data=Depr1
).fit(disp=False)


# Display the regression summary.

print("\nLOGISTIC REGRESSION SUMMARY")

print(fit.summary())


# ------------------------------------------------
# 2B. Coefficients and significance tests
# ------------------------------------------------

# The regression summary contains:
#
#   coef       Estimated regression coefficient
#   std err    Standard error
#   z          Wald z-statistic
#   P>|z|      Two-sided p-value
#
# The Wald statistic is:
#
#     z = estimated coefficient / standard error
#
# Under H0: beta_j = 0, the statistic has
# approximately a standard normal distribution.
#
# Small p-values indicate evidence that a predictor
# is associated with the response, conditional on
# the other predictors in the model.


coefficients = pd.DataFrame({
    "Estimate": fit.params,
    "Std. Error": fit.bse,
    "z value": fit.tvalues,
    "p-value": fit.pvalues
})

print("\nPARAMETER ESTIMATES")

print(coefficients.round(5))


# ------------------------------------------------
# 2C. Odds ratios
# ------------------------------------------------

# The exponential of a logistic regression
# coefficient is an odds ratio.
#
# For a one-unit increase in a numerical predictor,
# holding the other predictors constant:
#
#     Odds ratio = exp(beta_j)
#
# An odds ratio below 1 indicates decreasing odds
# of a positive diagnosis as the predictor increases.
#
# An odds ratio above 1 indicates increasing odds.
#
# For Gender, the odds ratio compares Male with
# the reference category Female.


odds_ratios = np.exp(fit.params)

print("\nODDS RATIOS")

print(odds_ratios.round(4))

# ================================================================
# PART 3. CLASSIFICATION AND TRAINING ERROR
# ================================================================

# We now evaluate how well the fitted model
# classifies observations in the training data.
#
# IMPORTANT:
# These observations were used to estimate the
# regression coefficients.
#
# Therefore, this is an evaluation of training
# accuracy, not cross-validation.
#
# A separate testing sample or cross-validation
# would be needed to evaluate out-of-sample accuracy.


# ------------------------------------------------
# 3A. Predicted probabilities
# ------------------------------------------------

# The predict() method returns fitted probabilities:
#
#     Prob_i = P(Diagnosis_i = 1 | predictors)
#
# Each probability is between 0 and 1.


Prob = fit.predict(Depr1)

print("\nSUMMARY OF PREDICTED PROBABILITIES")

print(Prob.describe())


# ------------------------------------------------
# 3B. Classify observations
# ------------------------------------------------

# We classify a student as having a positive
# diagnosis when the predicted probability
# exceeds 0.30.
#
# Otherwise, the predicted diagnosis is 0.
#
# The threshold 0.30 is chosen for illustration.
# It is not a validated clinical decision threshold.


YesPredict = (Prob > 0.30).astype(int)


# ------------------------------------------------
# 3C. Confusion matrix
# ------------------------------------------------

# A confusion matrix compares the actual and
# predicted classifications.
#
# Rows represent actual diagnoses.
# Columns represent predicted diagnoses.
#
# The four possible outcomes are:
#
#   True negative:  Actual 0, Predicted 0
#   False positive: Actual 0, Predicted 1
#   False negative: Actual 1, Predicted 0
#   True positive:  Actual 1, Predicted 1


table = pd.crosstab(
    Depr1["Diagnosis"],
    YesPredict,
    rownames=["Actual diagnosis"],
    colnames=["Predicted diagnosis"]
)

print("\nCONFUSION MATRIX")

print(table)


# ------------------------------------------------
# 3D. Classification accuracy
# ------------------------------------------------

# Accuracy is the proportion of correctly
# classified observations:
#
#     Accuracy = (TN + TP) / n
#
# Training error is:
#
#     Training error = 1 - Accuracy


accuracy = accuracy_score(
    Depr1["Diagnosis"],
    YesPredict
)

training_error = 1 - accuracy


print("\nCLASSIFICATION ACCURACY")

print("Accuracy:", round(accuracy, 4))

print("Training error:", round(training_error, 4))


# ------------------------------------------------
# 3E. Sensitivity and specificity
# ------------------------------------------------

# Overall accuracy can be misleading when one
# response category is much more common.
#
# Sensitivity measures the proportion of actual
# positive diagnoses correctly identified.
#
#     Sensitivity = TP / (TP + FN)
#
# Specificity measures the proportion of actual
# negative diagnoses correctly identified.
#
#     Specificity = TN / (TN + FP)


TN, FP, FN, TP = confusion_matrix(
    Depr1["Diagnosis"],
    YesPredict,
    labels=[0, 1]
).ravel()


sensitivity = TP / (TP + FN)

specificity = TN / (TN + FP)


print("\nSENSITIVITY AND SPECIFICITY")

print("Sensitivity:", round(sensitivity, 4))

print("Specificity:", round(specificity, 4))

# ================================================================
# PART 4. PREDICTION FOR NEW OBSERVATIONS
# ================================================================

# Logistic regression can predict the probability
# of a positive diagnosis for a new observation.
#
# First, the regression model calculates the logit:
#
#     eta = b0 + b1*x1 + b2*x2 + b3*x3
#
# The logit is converted into a probability using
# the logistic function:
#
#     p = exp(eta) / (1 + exp(eta))
#
# Equivalently:
#
#     p = 1 / (1 + exp(-eta))
#
# scipy.special.expit() calculates this probability
# accurately, including for very large logits.


# ------------------------------------------------
# 4A. Prediction for a student with low cohesion
# ------------------------------------------------

# Consider a female student with:
#
#   Gender = Female
#   Guardian_status = 1
#   Cohesion_score = 26
#
# This represents a student with a low
# community cohesion score.


student1 = pd.DataFrame({
    "Gender": ["Female"],
    "Guardian_status": [1],
    "Cohesion_score": [26]
})


# Predict the logit.
#
# linear=True returns the linear predictor
# rather than the probability.

logit1 = fit.predict(student1, which="linear").iloc[0]

print("\nPREDICTED LOGIT")

print(logit1)


# Convert the logit into a probability.

P1 = expit(logit1)

print("\nPREDICTED PROBABILITY")

print(P1)


# ------------------------------------------------
# 4B. Predict probability directly
# ------------------------------------------------

# By default, predict() returns the probability
# rather than the logit.
#
# This is equivalent to:
#
# predict(fit, newdata=student1, type="response")
#
# in R.


P1_direct = fit.predict(student1).iloc[0]

print("\nDIRECTLY PREDICTED PROBABILITY")

print(P1_direct)


# ------------------------------------------------
# 4C. Prediction with average cohesion
# ------------------------------------------------

# Now consider the same student, but with
# Cohesion_score = 52.
#
# All other predictors remain unchanged.


student2 = pd.DataFrame({
    "Gender": ["Female"],
    "Guardian_status": [1],
    "Cohesion_score": [52]
})


P2 = fit.predict(student2).iloc[0]


print("\nPREDICTION WITH HIGHER COHESION")

print("Cohesion score 26:", round(P1, 4))

print("Cohesion score 52:", round(P2, 4))

# ================================================================
# PART 5. VISUALIZATION
# ================================================================

# We can visualize how the predicted probability
# changes with the community cohesion score.
#
# We keep Gender = Female and Guardian_status = 1
# fixed while varying Cohesion_score.
#
# The resulting curve is nonlinear, even though
# Cohesion_score enters linearly in the logit.


import matplotlib.pyplot as plt


cohesion_values = np.linspace(
    Depr1["Cohesion_score"].min(),
    Depr1["Cohesion_score"].max(),
    200
)


new_students = pd.DataFrame({
    "Gender": ["Female"] * len(cohesion_values),
    "Guardian_status": [1] * len(cohesion_values),
    "Cohesion_score": cohesion_values
})


probabilities = fit.predict(new_students)


plt.figure(figsize=(8, 5))

plt.plot(
    cohesion_values,
    probabilities
)

plt.scatter(
    [26, 52],
    [P1, P2],
    color="red",
    zorder=3
)

plt.xlabel("Community cohesion score")

plt.ylabel("Predicted probability of positive diagnosis")

plt.title("Logistic Regression: Predicted Probability")

plt.ylim(0, 1)

plt.grid(alpha=0.3)

plt.show()
