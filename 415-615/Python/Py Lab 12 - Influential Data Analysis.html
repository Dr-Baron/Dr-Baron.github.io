# ================================================================
# PART 1. LEVERAGE AND THE HAT MATRIX
# ================================================================

# An observation may have an unusual combination of
# predictor values, even if its response is not unusual.
#
# Such observations are said to have high leverage.
#
# The hat matrix is:
#
#     H = X (X'X)^(-1) X'
#
# The diagonal elements Hii measure leverage.
#
# We do not need to calculate the entire hat matrix.
# statsmodels provides its diagonal directly.


# ------------------------------------------------
# 1A. Obtain the influence diagnostics
# ------------------------------------------------

# get_influence() calculates regression diagnostics
# for individual observations.
#
# This is similar to influence(reg) in R.

infl = reg.get_influence()


# Examine the leverage values.

leverage = infl.hat_matrix_diag

print("\nLEVERAGE VALUES")

print(
    pd.Series(
        leverage,
        index=mtcars.index,
        name="Leverage"
    ).round(4)
)


# ------------------------------------------------
# 1B. Mean leverage
# ------------------------------------------------

# In multiple regression with an intercept:
#
#     sum(Hii) = p + 1
#
# where p is the number of predictors.
#
# Therefore:
#
#     mean(Hii) = (p + 1) / n
#
# For mtcars:
#     p = 10 predictors
#     n = 32 observations
#
# Expected mean leverage = 11/32 = 0.34375.


n = len(Y)

p = X.shape[1] - 1

mean_leverage = (p + 1) / n

print("\nTHEORETICAL MEAN LEVERAGE:")
print(mean_leverage)

print("\nACTUAL MEAN LEVERAGE:")
print(np.mean(leverage))


# ------------------------------------------------
# 1C. Plot leverage values
# ------------------------------------------------

# Plot Hii against observation number.
#
# Observations with unusually large leverage
# may deserve further investigation.
#
# A commonly used screening threshold is
# twice the average leverage:
#
#     Hii > 2(p + 1)/n
#
# This is a rule of thumb, not a formal test.


plt.figure(figsize=(9, 5))

plt.scatter(
    range(1, n + 1),
    leverage
)

plt.axhline(
    mean_leverage,
    color="gray",
    linestyle="--",
    label="Mean leverage"
)

plt.axhline(
    2 * mean_leverage,
    color="red",
    linestyle="--",
    label="Twice mean leverage"
)

plt.xlabel("Observation number")
plt.ylabel("Leverage")
plt.title("Leverage Values")

plt.legend()
plt.grid(alpha=0.3)

plt.show()

# ================================================================
# PART 2. DELETING INDIVIDUAL OBSERVATIONS
# ================================================================

# An observation is influential if removing it
# substantially changes the fitted regression model.
#
# We can examine what happens when each observation
# is deleted from the dataset.
#
# The influence diagnostics provide:
#
#   1. Regression coefficients without observation i.
#   2. Residual variance without observation i.
#
# These correspond to the quantities:
#
#     infl$coefficients
#     infl$sigma
#
# in the R lab.


# ------------------------------------------------
# 2A. Regression coefficients after deletion
# ------------------------------------------------

# params_not_obsi contains the regression coefficient
# estimates after deleting each observation.
#
# Each row corresponds to one deleted observation.
#
# Each column corresponds to a regression coefficient.


coefficients_deleted = pd.DataFrame(
    infl.params_not_obsi,
    index=mtcars.index,
    columns=X.columns
)

print("\nREGRESSION COEFFICIENTS AFTER DELETION")

print(coefficients_deleted.round(4))


# ------------------------------------------------
# 2B. Residual standard deviations after deletion
# ------------------------------------------------

# sigma2_not_obsi contains the estimated residual
# variance after deleting each observation.
#
# Taking the square root gives the residual
# standard deviation, or RMSE, of each reduced model.


sigma_deleted = pd.Series(
    np.sqrt(infl.sigma2_not_obsi),
    index=mtcars.index,
    name="RMSE after deletion"
)

print("\nRESIDUAL STANDARD DEVIATIONS AFTER DELETION")

print(sigma_deleted.round(4))

# ================================================================
# PART 3. OUTLIER DETECTION
# ================================================================

# An outlier is an observation with an unusually
# large residual relative to its estimated variability.
#
# Externally studentized residuals are especially useful
# for detecting outliers.
#
# The externally studentized residual for observation i
# uses the residual variance estimated without
# observation i.
#
# Under the normal linear regression model, these
# residuals have a t-distribution with
#
#     n - p - 2
#
# degrees of freedom, where p is the number of slopes.
#
# We can use them to test whether an individual
# observation is an outlier.


# ------------------------------------------------
# 3A. Externally studentized residuals
# ------------------------------------------------

rstudent = infl.resid_studentized_external

print("\nEXTERNALLY STUDENTIZED RESIDUALS")

print(
    pd.Series(
        rstudent,
        index=mtcars.index,
        name="RStudent"
    ).round(4)
)


# ------------------------------------------------
# 3B. Outlier tests
# ------------------------------------------------

# For each observation, test:
#
#     H0: The observation follows the regression model.
#
# A two-sided p-value is calculated from the
# externally studentized residual.
#
# Since we test all n observations, we also calculate
# Bonferroni-adjusted p-values:
#
#     adjusted p-value = min(n * p-value, 1)
#
# This corresponds to outlierTest(reg) from R's
# car package.


df = int(reg.df_resid - 1)

pvalues = 2 * t.sf(
    np.abs(rstudent),
    df
)

pvalues_bonf = np.minimum(
    n * pvalues,
    1
)


outliers = pd.DataFrame({
    "RStudent": rstudent,
    "Unadjusted p": pvalues,
    "Bonferroni p": pvalues_bonf
}, index=mtcars.index)


# Display the observations with the smallest p-values.

print("\nOUTLIER TESTS")

print(
    outliers.sort_values("Unadjusted p").head(10)
)


# Display observations significant after Bonferroni
# correction at the 5% level.

print("\nBONFERRONI-SIGNIFICANT OUTLIERS")

print(
    outliers[
        outliers["Bonferroni p"] < 0.05
    ]
)

# ================================================================
# PART 4. COOK'S DISTANCE
# ================================================================

# Cook's distance measures the overall influence
# of an observation on the fitted regression model.
#
# It combines information about:
#
#   1. The size of the residual.
#   2. The leverage of the observation.
#
# A large Cook's distance indicates that deleting
# the observation substantially changes the fitted
# regression values.
#
# Unlike leverage alone, Cook's distance measures
# the combined effect of unusual predictors and
# an unusual response.

# 4A. Calculate Cook's distance
# ------------------------------------------------

# cooks_distance returns Cook's distances and
# their associated reference p-values.
#

# ================================================================
# PART 5. ADDITIONAL INFLUENCE MEASURES
# ================================================================

# Cook's distance summarizes the overall effect
# of deleting an observation.
#
# Other diagnostics examine particular aspects
# of that effect.
#
# DFBETAS:
#   Measures how much each regression coefficient
#   changes when observation i is deleted,
#   standardized by its estimated standard error.
#
# DFFITS:
#   Measures the change in the fitted value for
#   observation i when that observation is deleted.
#
# COVRATIO:
#   Measures the effect of deleting an observation
#   on the generalized variance of the estimated
#   regression coefficients.
#
# statsmodels calculates all of these quantities.


# ------------------------------------------------
# 5A. DFBETAS
# ------------------------------------------------

# There is one DFBETAS value for every observation
# and every regression coefficient.
#
# Large absolute values indicate that the observation
# has a substantial effect on that coefficient.


dfbetas = pd.DataFrame(
    infl.dfbetas,
    index=mtcars.index,
    columns=X.columns
)

print("\nDFBETAS")

print(dfbetas.round(3))


# ------------------------------------------------
# 5B. DFFITS
# ------------------------------------------------

# DFFITS measures the standardized change in the
# fitted value when an observation is removed.

dffits = pd.Series(
    infl.dffits[0],
    index=mtcars.index,

# This section saves the current Google Colab notebook
# in two formats:
#
# HTML:
#   A readable document containing the Python code,
#   explanatory comments, tables, and graphical output.
#
# IPYNB:
#   An executable Jupyter notebook that can be opened
#   and edited in Google Colab or Jupyter.
#
# Run this section AFTER executing the previous cells,
# so that their output is included in the exported files.
#
# Both files will be downloaded to your computer.
# Your browser's download settings determine where
# they are saved.


from google.colab import _message, files
from nbconvert import HTMLExporter
import nbformat
import json


# Retrieve the current Colab notebook.

notebook = _message.blocking_request(
    "get_ipynb",
    timeout_sec=60
)["ipynb"]


# Convert to the standard Jupyter notebook format.

nb = nbformat.reads(
    notebook if isinstance(notebook, str)
    else json.dumps(notebook),
    as_version=4
)


# Specify a common filename for both versions.

filename = "Py Lab 12 - Influential Data Analysis"


# ------------------------------------------------
# Save the executable Jupyter notebook
# ------------------------------------------------

with open(filename + ".ipynb", "w", encoding="utf-8") as f:
    nbformat.write(nb, f)


# ------------------------------------------------
# Convert the notebook to HTML
# ------------------------------------------------

html, resources = HTMLExporter().from_notebook_node(nb)


# Save the HTML document.

with open(filename + ".html", "w", encoding="utf-8") as f:
    f.write(html)


# ------------------------------------------------
# Download both files
# ------------------------------------------------

files.download(filename + ".html")

files.download(filename + ".ipynb")
