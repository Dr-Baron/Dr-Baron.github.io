# ================================================================
# PART 1. FITTING A NONLINEAR REGRESSION MODEL
# ================================================================

# We want to predict fuel economy using car weight
# and the number of cylinders.
#
# Consider the nonlinear regression model:
#
# log(mpg) = b0 + b1 * exp(b2 * wt + cyl) + error
#
# Here:
#
#   b0, b1, b2 are unknown regression parameters.
#   wt is the automobile's weight.
#   cyl is the number of cylinders.
#
# The model is nonlinear in the parameter b2.
#
# Therefore, ordinary linear regression cannot be
# used to estimate all three parameters.
#
# Instead, we minimize the residual sum of squares
# numerically using nonlinear least squares.


# ------------------------------------------------
# 1A. Define the nonlinear regression function
# ------------------------------------------------

# The function takes two predictors and three parameters.
#
# curve_fit() will estimate b0, b1, and b2.

def nonlinear_model(X, b0, b1, b2):

    wt, cyl = X

    return b0 + b1 * np.exp(b2 * wt + cyl)


# Define the response and predictors.

Y = np.log(mtcars["mpg"].to_numpy())

wt = mtcars["wt"].to_numpy()

cyl = mtcars["cyl"].to_numpy()


# ------------------------------------------------
# 1B. Estimate the parameters
# ------------------------------------------------

# Nonlinear least squares requires initial estimates
# of the unknown parameters.
#
# Unlike linear regression, the numerical algorithm
# may converge to different solutions depending on
# the initial estimates.
#
# We use starting values near those obtained in the
# original R lab to facilitate convergence.
#
# The parameter estimates are obtained by minimizing:
#
# SSE = sum((Yi - f(Xi; b0, b1, b2))^2)


initial = [3.2, -0.00005, 0.29]

params, covariance = curve_fit(
    nonlinear_model,
    (wt, cyl),
    Y,
    p0=initial,
    maxfev=10000
)


# Extract the estimated parameters.

b0, b1, b2 = params

print("\nNONLINEAR REGRESSION")

print("b0 =", b0)
print("b1 =", b1)
print("b2 =", b2)


# ------------------------------------------------
# 1C. Residual sum of squares
# ------------------------------------------------

# Calculate the fitted values of log(mpg).

Yhat = nonlinear_model(
    (wt, cyl),
    b0, b1, b2
)


# Calculate the residuals.

residuals = Y - Yhat


# Calculate the residual sum of squares.

SSE = np.sum(residuals ** 2)

print("\nResidual sum of squares:", round(SSE, 4))


# The residual degrees of freedom equal:
#
#     n - k
#
# where k = 3 is the number of estimated parameters.

n = len(Y)

k = len(params)

df = n - k


# Residual standard error.

RSE = np.sqrt(SSE / df)

print("Residual standard error:", round(RSE, 4))
print("Residual degrees of freedom:", df)

# PART 2. PARAMETER ESTIMATES AND SIGNIFICANCE TESTS
# ================================================================

# The covariance matrix returned by curve_fit()
# provides approximate variances and covariances
# of the estimated regression parameters.
#
# The standard errors are the square roots of
# its diagonal elements.
#
# For each parameter, we test:
#
#     H0: beta_j = 0
#
# using the approximate t-statistic:
#
#              estimated beta_j
#     t = ----------------------------
#          SE(estimated beta_j)
#
# The tests use n - k residual degrees of freedom.
#
# These are approximate tests based on the local
# linearization of the nonlinear regression model.


# ------------------------------------------------
# 2A. Standard errors
# ------------------------------------------------

SE = np.sqrt(np.diag(covariance))


# ------------------------------------------------
# 2B. t-statistics and p-values
# ------------------------------------------------

tvalues = params / SE

pvalues = 2 * t.sf(
    np.abs(tvalues),
    df
)


# ------------------------------------------------
# 2C. Display the regression summary
# ------------------------------------------------

summary = pd.DataFrame({
    "Estimate": params,
    "Std. Error": SE,
    "t value": tvalues,
    "p-value": pvalues
}, index=["b0", "b1", "b2"])

print("\nPARAMETER ESTIMATES")

print(summary.round(6))

# ================================================================
# PART 3. PREDICTION
# ================================================================

# Predicted values are calculated by substituting
# the estimated parameters into the regression function.
#
# Our model predicts log(mpg).
#
# Therefore:
#
#     log(mpg)_predicted = b0 + b1 * exp(b2*wt + cyl)
#
# Exponentiating gives predictions on the mpg scale.
#
# Note that exp(predicted log(mpg)) is not generally
# an unbiased estimate of the conditional mean mpg.
# It is the exponentiated fitted log response.


# ------------------------------------------------
# 3A. Predicted log(mpg)
# ------------------------------------------------

log_mpg_predicted = nonlinear_model(
    (wt, cyl),
    *params
)


# ------------------------------------------------
# 3B. Predicted mpg
# ------------------------------------------------

mpg_predicted = np.exp(log_mpg_predicted)


# Display observed and predicted values.

predictions = pd.DataFrame({
    "Observed mpg": mtcars["mpg"],
    "Predicted log(mpg)": log_mpg_predicted,
    "Predicted mpg": mpg_predicted
})

print("\nOBSERVED AND PREDICTED VALUES")

print(predictions.round(3))

# This section saves the current Google Colab notebook
# in two formats:
#
# HTML:
#   A readable document containing the Python code,
#   explanatory comments, tables, and graphical output.
#
# IPYNB:
#   An executable notebook that can be opened and
#   edited in Google Colab or Jupyter.
#
# Run this section AFTER executing the previous cells.
#
# Both files will be downloaded to your computer.
# Your browser's download settings determine the folder
# where the files are saved.


from google.colab import _message, files
from nbconvert import HTMLExporter
import nbformat
import json


# Retrieve the current Google Colab notebook.

notebook = _message.blocking_request(
    "get_ipynb",
    timeout_sec=60
)["ipynb"]


# Convert it to standard Jupyter notebook format.

nb = nbformat.reads(
    notebook if isinstance(notebook, str)
    else json.dumps(notebook),
    as_version=4
)


# Specify the common filename.

filename = "Py Lab 13 - Nonlinear Regression"


# ------------------------------------------------
# Save the executable Jupyter notebook
# ------------------------------------------------

with open(filename + ".ipynb", "w", encoding="utf-8") as f:
    nbformat.write(nb, f)


# ------------------------------------------------
# Convert the notebook to HTML
# ------------------------------------------------

html, resources = HTMLExporter().from_notebook_node(nb)


# Save the HTML document.

with open(filename + ".html", "w", encoding="utf-8") as f:
    f.write(html)


# ------------------------------------------------
# Download both files
# ------------------------------------------------

files.download(filename + ".html")

files.download(filename + ".ipynb")
