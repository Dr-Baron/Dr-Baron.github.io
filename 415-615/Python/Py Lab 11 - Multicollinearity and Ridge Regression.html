# ================================================================
# PART 1. MEASURING MULTICOLLINEARITY
# ================================================================

# Multicollinearity occurs when two or more predictors
# are strongly related to each other.
#
# It can make individual regression coefficients unstable
# and increase their standard errors.
#
# A useful first step is to examine correlations
# among the variables.


# ------------------------------------------------
# 1A. Correlation matrix
# ------------------------------------------------

# corr() calculates the Pearson correlation coefficient
# for every pair of numerical variables.
#
# Correlations close to +1 or -1 indicate strong
# linear relationships.

print("\nCORRELATION MATRIX")

print(mtcars.corr().round(3))


# ------------------------------------------------
# 1B. Variance inflation factors
# ------------------------------------------------

# Variance inflation factors (VIF) measure how strongly
# each predictor is linearly related to the other predictors.
#
# For predictor Xj:
#
#                1
#     VIFj = ----------
#             1 - Rj^2
#
# where Rj^2 is obtained by regressing Xj on all the
# other predictors.
#
# A VIF of 1 indicates no linear association with
# the other predictors.
#
# Large VIF values indicate substantial multicollinearity.
#
# Values above 5 or 10 are often used as informal
# warning thresholds, not strict statistical rules.


# Fit the full regression:
#
# mpg = beta0 + beta1*cyl + ... + beta10*carb + error

Y = mtcars["mpg"]

X = mtcars.drop(columns="mpg")

X1 = sm.add_constant(X)

reg = sm.OLS(Y, X1).fit()


# Calculate VIF for each predictor.
#
# The intercept is excluded from the VIF table.

vif = pd.Series(
    [
        variance_inflation_factor(X1.values, j)
        for j in range(1, X1.shape[1])
    ],
    index=X.columns,
    name="VIF"
)

print("\nVARIANCE INFLATION FACTORS")

print(vif.round(3))


# ------------------------------------------------
# 1C. Generalized VIF for categorical predictors
# ------------------------------------------------

# When a categorical predictor has several categories,
# it is represented by multiple dummy variables.
#
# For example, a variable with three categories requires
# two dummy variables.
#
# Instead of calculating separate VIF values for these
# dummy variables, we can calculate a generalized VIF
# (GVIF) for the entire categorical term.
#
# The adjusted quantity
#
#           GVIF^(1/(2*Df))
#
# accounts for the number of coefficients (Df)
# associated with the term.
#
# For a one-coefficient term, GVIF equals the ordinary VIF.
#
# We use the HOME_SALES dataset from the previous labs.


HOMES = pd.read_csv(
    "https://dr-baron.github.io/415-615/Data/HOME_SALES.csv"
)


# Exclude the response and the observation identifier.

Yhome = HOMES["SALES_PRICE"]

Xhome = HOMES.drop(columns=["ID", "SALES_PRICE"])


# STYLE is a categorical variable even though its
# categories are represented by numbers.

Xhome["STYLE"] = Xhome["STYLE"].astype("category")


# Convert categorical predictors to dummy variables.
#
# drop_first=True removes one category from each group,
# preventing perfect multicollinearity with the intercept.

Xhome = pd.get_dummies(
    Xhome,
    drop_first=True,
    dtype=float
)


# Fit the full regression model.

Xhome1 = sm.add_constant(Xhome)

reg_home = sm.OLS(Yhome, Xhome1).fit()


# ------------------------------------------------
# Calculate generalized VIF
# ------------------------------------------------

# GVIF can be calculated using the correlation matrix
# of the estimated regression coefficients.
#
# For a term represented by a group of coefficients,
#
#                  det(R_term) det(R_other)
#     GVIF = --------------------------------
#                         det(R)
#
# where R is the correlation matrix of all slope
# estimates, excluding the intercept.
#
# For one-coefficient terms, this gives the usual VIF.
#
# The following function groups dummy variables belonging
# to the same original categorical predictor.


def generalized_vif(regression, groups):

    # Covariance matrix of the slope estimates,
    # excluding the intercept.

    V = regression.cov_params().iloc[1:, 1:]

    # Convert covariance matrix to correlation matrix.

    sd = np.sqrt(np.diag(V))

    R = V.div(sd, axis=0).div(sd, axis=1).to_numpy()

    names = list(V.index)

    # Use log-determinants for numerical stability.

    _, logdet_R = np.linalg.slogdet(R)

    results = []

    for term, columns in groups.items():

        idx = [names.index(c) for c in columns]

        other = [
            j for j in range(len(names))
            if j not in idx
        ]

        R_term = R[np.ix_(idx, idx)]
        R_other = R[np.ix_(other, other)]

        _, logdet_term = np.linalg.slogdet(R_term)

        logdet_other = (
            np.linalg.slogdet(R_other)[1]
            if other else 0
        )

        GVIF = np.exp(
            logdet_term + logdet_other - logdet_R
        )

        df = len(idx)

        results.append({
            "Term": term,
            "GVIF": GVIF,
            "Df": df,
            "GVIF^(1/(2*Df))": GVIF ** (1 / (2 * df))
        })

    return pd.DataFrame(results).set_index("Term")


# Group the dummy-variable columns by original predictor.
#
# For example, QUALITY_LOW and QUALITY_MEDIUM
# belong to the same term, QUALITY.

groups = {}

for col in HOMES.drop(columns=["ID", "SALES_PRICE"]).columns:

    if col in Xhome.columns:
        groups[col] = [col]

    else:
        groups[col] = [
            c for c in Xhome.columns
            if c.startswith(col + "_")
        ]


# Calculate generalized VIF.

gvif = generalized_vif(reg_home, groups)

print("\nGENERALIZED VIF")

print(gvif.round(3))

# ================================================================
# PART 2. RIDGE REGRESSION
# ================================================================

# Ridge regression is a method for reducing the instability
# of regression coefficients caused by multicollinearity.
#
# Ordinary least squares minimizes:
#
#     SSE = sum((Yi - Yhat_i)^2)
#
# Ridge regression minimizes:
#
#     SSE + lambda * sum(beta_j^2)
#
# where the penalty applies to the slopes, not the intercept.
#
# lambda >= 0 controls the amount of shrinkage.
#
# lambda = 0:
#     Ridge regression reduces to ordinary least squares.
#
# Larger lambda:
#     Coefficients are increasingly shrunk toward zero.
#
# Ridge regression generally does not set coefficients
# exactly equal to zero.
#
# IMPORTANT:
# Predictor variables should usually be standardized
# before ridge regression because they may be measured
# on very different scales.
#
# sklearn uses the penalty parameter alpha for lambda.
# Here we will call it lambda in our comments and graphs.


# ------------------------------------------------
# 2A. Standardize the predictors
# ------------------------------------------------

# We again use mtcars, predicting mpg from all
# the other variables.
#
# StandardScaler subtracts each predictor's mean
# and divides by its standard deviation.
#
# This makes the penalty comparable across predictors.

Y = mtcars["mpg"]

X = mtcars.drop(columns="mpg")

scaler = StandardScaler()

Xs = scaler.fit_transform(X)


# ------------------------------------------------
# 2B. Fit ridge regression with lambda = 0.5
# ------------------------------------------------

# Ridge() fits the regression model for a specified
# penalty parameter.
#
# The intercept is not penalized.

rr = Ridge(alpha=0.5)

rr.fit(Xs, Y)


# Display the estimated coefficients.
#
# These are coefficients for standardized predictors.

coef = pd.Series(
    rr.coef_,
    index=X.columns
)

print("\nRIDGE REGRESSION: lambda = 0.5")

print("Intercept:", round(rr.intercept_, 4))

print("\nCoefficients for standardized predictors:")
print(coef.round(4))


# ------------------------------------------------
# 2C. Ridge regression for a range of lambda values
# ------------------------------------------------

# Instead of choosing one lambda, we can fit ridge
# regression for many values.
#
# We will use lambda = 0, 0.1, 0.2, ..., 30.
#
# At lambda = 0, the fitted model is ordinary
# least squares.
#
# As lambda increases, the standardized coefficients
# are gradually shrunk toward zero.


lambdas = np.arange(0, 30.1, 0.1)

coefficients = []

for lam in lambdas:

    rr = Ridge(alpha=lam)

    rr.fit(Xs, Y)

    coefficients.append(rr.coef_)


coefficients = np.array(coefficients)


# ------------------------------------------------
# 2D. Plot the ridge coefficient paths
# ------------------------------------------------

# Each curve shows how one regression coefficient
# changes as lambda increases.
#
# Coefficients are expressed on the standardized
# predictor scale.

plt.figure(figsize=(10, 6))

for j, variable in enumerate(X.columns):

    plt.plot(
        lambdas,
        coefficients[:, j],
        label=variable
    )

plt.axhline(0, color="black", linewidth=0.7)

plt.xlabel("Lambda")
plt.ylabel("Ridge regression coefficient")
plt.title("Ridge Coefficient Paths")

plt.legend()
plt.grid(alpha=0.3)

plt.show()


# ------------------------------------------------
# 2E. Generalized cross-validation (GCV)
# ------------------------------------------------

# We need a method for choosing lambda.
#
# Generalized cross-validation estimates prediction
# error without repeatedly fitting models on
# different training and testing samples.
#
# For ridge regression:
#
#                    SSE(lambda) / n
#     GCV(lambda) = ----------------------
#                    (1 - df(lambda)/n)^2
#
# where df(lambda) is the effective number of
# fitted parameters.
#
# Unlike ordinary least squares, ridge regression
# has a fractional effective degrees of freedom.
#
# Smaller GCV indicates better estimated
# prediction performance.
#
# We standardize the response for this calculation
# to make the GCV values dimensionless.
#
# The following calculation uses the singular values
# of the standardized design matrix.


# Center and standardize the response.

ys = (
    Y.to_numpy() - Y.mean()
) / Y.std(ddof=0)


# Singular value decomposition of Xs.

U, singular_values, Vt = np.linalg.svd(
    Xs,
    full_matrices=False
)


n = len(Y)

GCV = []

for lam in lambdas:

    # Fit ridge regression to the standardized response.

    rr = Ridge(alpha=lam)

    rr.fit(Xs, ys)

    # Sum of squared residuals.

    residuals = ys - rr.predict(Xs)

    SSE = np.sum(residuals ** 2)

    # Effective degrees of freedom for ridge regression.
    #
    # Add 1 for the unpenalized intercept.

    df = 1 + np.sum(
        singular_values ** 2 /
        (singular_values ** 2 + lam)
    )

    # Generalized cross-validation.

    gcv = (SSE / n) / (1 - df / n) ** 2

    GCV.append(gcv)


GCV = np.array(GCV)


# ------------------------------------------------
# 2F. Plot GCV against lambda
# ------------------------------------------------

plt.figure(figsize=(9, 5))

plt.plot(
    lambdas,
    GCV,
    marker=".",
    markersize=3
)

plt.xlabel("Lambda")
plt.ylabel("GCV")
plt.title("Generalized Cross-Validation")

plt.grid(alpha=0.3)

plt.show()


# ------------------------------------------------
# 2G. Find the optimal lambda
# ------------------------------------------------

# Choose the lambda value that minimizes GCV.

lambda_optimal = lambdas[np.argmin(GCV)]

print("\nOPTIMAL LAMBDA")

print("Lambda minimizing GCV:",
      round(lambda_optimal, 3))

print("Minimum GCV:",
      round(GCV.min(), 4))


# ------------------------------------------------
# 2H. Fit the optimal ridge regression model
# ------------------------------------------------

# Fit the ridge regression using the optimal lambda.

rr_optimal = Ridge(alpha=lambda_optimal)

rr_optimal.fit(Xs, Y)


# Display the intercept and coefficients.
#
# The following coefficients are on the standardized
# predictor scale.

print("\nOPTIMAL RIDGE REGRESSION")

print("Intercept:", round(rr_optimal.intercept_, 4))

print("\nCoefficients:")

print(
    pd.Series(
        rr_optimal.coef_,
        index=X.columns
    ).round(4)
)


# ------------------------------------------------
# 2I. Prediction with ridge regression
# ------------------------------------------------

# Ridge regression is still a linear regression model.
#
# Predicted values are calculated by:
#
#     Yhat = beta0 + X * beta
#
# sklearn provides the predict() function, so we
# do not need to perform the matrix multiplication
# manually.
#
# We calculate predicted mpg for the observations
# in the mtcars dataset.


mpg_predicted = rr_optimal.predict(Xs)


# Display observed and predicted values.

prediction_table = pd.DataFrame({
    "Observed mpg": Y,
    "Predicted mpg": mpg_predicted
})

print("\nOBSERVED AND PREDICTED MPG")

print(prediction_table.round(2))


# ------------------------------------------------
# 2J. Plot observed versus predicted mpg
# ------------------------------------------------

# If the predictions were perfect, all points would
# lie on the diagonal line Yhat = Y.
#
# The closer the points are to this line,
# the better the fit to the observed data.

plt.figure(figsize=(7, 6))

plt.scatter(
    Y,
    mpg_predicted
)

limits = [
    min(Y.min(), mpg_predicted.min()),
    max(Y.max(), mpg_predicted.max())
]

plt.plot(
    limits,
    limits,
    "k--"
)

plt.xlabel("Observed mpg")
plt.ylabel("Predicted mpg")
plt.title("Ridge Regression: Observed vs Predicted")

plt.grid(alpha=0.3)

plt.show()


# ------------------------------------------------
# 2K. Calculate predictions using matrix multiplication
# ------------------------------------------------

# To illustrate the linear regression formula,
# we can calculate the same predictions manually.
#
# Ridge.predict() uses:
#
#     Yhat = beta0 + Xs @ beta
#
# where @ denotes matrix multiplication in Python.


mpg_predicted_manual = (
    rr_optimal.intercept_
    + Xs @ rr_optimal.coef_
)


# Verify that both methods give the same predictions.

print("\nMAXIMUM DIFFERENCE BETWEEN PREDICTIONS")

print(
    np.max(
        np.abs(
            mpg_predicted - mpg_predicted_manual
        )
    )
)
